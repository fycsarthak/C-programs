Loops

Program 1: Print numbers from 1 to 5

Code:

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 5; i++)
    {
        printf("%d ", i);
    }

    return 0;
}

Output:

1 2 3 4 5

Arrays

Program 1: Print array elements

Code:

#include <stdio.h>

int main()
{
    int a[3] = {10, 20, 30};
    int i;

    for(i = 0; i < 3; i++)
    {
        printf("%d ", a[i]);
    }

    return 0;
}

Output:

10 20 30
