{
printf("Check the dates entered.\n");
return 0;

if (cy % 400 == 0 || (cy % 4 == 0 && cy % 100 != 0)) mdays[2] = 29;
if (cd < bd)
{

cm--;
if (cm == 0) { cm = 12; cy--; }
days = mdays[cm];
if (cm == 2 && (cy % 400 == 0 || (cy % 4 == 0 && cy % 100 != 0))) days = 29;

1. Convert a non-negative integer to binary.

#include <stdio.h>
int main(void)
{
int n, bits[32], count = 0, i;
printf("Enter a non-negative number: ");
scanf("%d", &n);
if (n == 0) printf("Binary = 0\n");
else if (n < 0) printf("Please enter a non-negative number.\n");
else
{
while (n > 0)
{
bits[count] = n % 2;
count++;
n = n / 2;
}
printf("Binary = ");
for (i = count - 1; i >= 0; i--) printf("%d", bits[i]);
printf("\n");
}
return 0;
}

Binary = 1101
